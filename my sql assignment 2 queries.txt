
-- MYSQL ASSIGNMENT 2 - QUERYING DATA

USE employee;

-- 1. DISTINCT
SELECT DISTINCT salary
FROM Employees;

-- 2. ALIAS
SELECT age AS Employee_Age,
       salary AS Employee_Salary
FROM Employees;

-- 3(a). WHERE CLAUSE
SELECT *
FROM Employees
WHERE salary > 50000
AND hire_date < '2016-01-01';

-- 3(b). UPDATE MISSING DESIGNATION
UPDATE Employees
SET designation = 'Data Scientist'
WHERE designation IS NULL;

-- 4. ORDER BY
SELECT *
FROM Employees
ORDER BY department_id ASC, salary DESC;

-- 5. LIMIT
SELECT *
FROM Employees
WHERE hire_date >= '2018-01-01'
AND hire_date < '2019-01-01'
ORDER BY hire_date ASC
LIMIT 5;

-- 6(a). SUM OF FINANCE SALARY
SELECT SUM(e.salary) AS Total_Finance_Salary
FROM Employees e
JOIN Departments d
ON e.department_id = d.department_id
WHERE d.department_name = 'Finance';

-- 6(b). MINIMUM AGE
SELECT MIN(age) AS Minimum_Age
FROM Employees;

-- 7(a). MAXIMUM SALARY BY LOCATION
SELECT l.location, MAX(e.salary) AS Maximum_Salary
FROM Employees e
JOIN Location l
ON e.location_id = l.location_id
GROUP BY l.location;

-- 7(b). AVERAGE SALARY FOR ANALYST DESIGNATIONS
SELECT designation, AVG(salary) AS Average_Salary
FROM Employees
WHERE designation LIKE '%Analyst%'
GROUP BY designation;

-- 8(a). DEPARTMENTS WITH LESS THAN 3 EMPLOYEES
SELECT d.department_name,
COUNT(e.employee_id) AS Employee_Count
FROM Departments d
LEFT JOIN Employees e
ON d.department_id = e.department_id
GROUP BY d.department_id, d.department_name
HAVING COUNT(e.employee_id) < 3;

-- 8(b). FEMALE EMPLOYEES BY LOCATION
SELECT l.location, AVG(e.age) AS Average_Age
FROM Location l
JOIN Employees e
ON l.location_id = e.location_id
WHERE e.gender = 'F'
GROUP BY l.location
HAVING AVG(e.age) < 30;

-- 9(a). INNER JOIN
SELECT e.employee_name, e.designation, d.department_name
FROM Employees e
INNER JOIN Departments d
ON e.department_id = d.department_id;

-- 9(b). LEFT JOIN
SELECT d.department_name,
COUNT(e.employee_id) AS Total_Employees
FROM Departments d
LEFT JOIN Employees e
ON d.department_id = e.department_id
GROUP BY d.department_id, d.department_name;

-- 9(c). RIGHT JOIN
SELECT l.location, e.employee_name
FROM Employees e
RIGHT JOIN Location l
ON e.location_id = l.location_id;

-- FINAL VERIFICATION
SELECT COUNT(*) FROM Employees;
